## 核心目标

## 6.1 评估集数据结构设计

```python
# ============================================================
# 建议的评估集数据结构（可以存成 JSON / YAML / Python dict）
# ============================================================
from dataclasses import dataclass, field
from typing import Literal


QuestionType = Literal["yes_no", "list", "how_to", "frequency", "fact"]
DifficultyTag = Literal[
    "normal_hit",    # 正常命中
    "weak_evidence", # 弱证据
    "hallucination_prone", # 容易幻觉
    "bad_synthesis", # 检索到了但生成不好
]


@dataclass
class RAGEvalCase:
    """
    单条 RAG 评估 Case。

    字段说明：
      case_id        : 唯一标识，方便定位
      question_type  : 问题类型（yes_no/list/how_to/frequency/fact）
      difficulty     : 难度标签（对应上面的四种典型场景）
      question       : 用户实际会问的问题
      expected_exact : 期望答案中必须出现的关键信息（出现才算答对）
      expected_absent: 期望答案中绝对不能出现的信息（出现就算幻觉）
      should_refuse  : True 表示模型应该拒答（无证据或弱证据场景）
      reference_chunks: 理想情况下应该命中的关键词（用于验证检索）
      notes          : 备注，记录这个 case 为什么难 / 容易出什么问题
    """
    case_id: str
    question_type: QuestionType
    difficulty: DifficultyTag
    question: str
    expected_exact: list[str] = field(default_factory=list)
    expected_absent: list[str] = field(default_factory=list)
    should_refuse: bool = False
    reference_chunks: list[str] = field(default_factory=list)
    notes: str = ""


@dataclass
class RAGEvalResult:
    """
    单条 Case 的评估结果。

    判断维度（对应任务.md 里的 5 个维度）：
      irrelevant     : 是否答非所问
      hallucination  : 是否幻觉（出现了 expected_absent 的词）
      wrong_refuse   : 是否该拒答却没拒（或不该拒答却拒了）
      citation_messy : 是否引用不清晰
      synthesis_bad  : 是否多 chunk 整合很乱
    """
    case_id: str
    model_output: str = ""
    irrelevant: bool = False
    hallucination: bool = False
    wrong_refuse: bool = False
    citation_messy: bool = False
    synthesis_bad: bool = False
    extra_notes: str = ""
```

## 6.3 评估脚本（最小可运行版本）

```python
"""
最小 RAG 评估脚本。

用法：
    1. 把上面 20 个 case 放在同一个 Python 文件
    2. 实现 run_rag_query(question) -> answer_text 函数（调用你的 LangChain RAG）
    3. 运行 eval_runner()，逐条输出结果
"""

import json
from typing import Callable


def run_rag_query(question: str) -> str:

    这里只是占位，实际项目中应该调用：
        stream_answer_with_knowledge_langchain_native(...)
    并收集最终的 answer_text。
    """

def check_case(case: RAGEvalCase, answer: str) -> RAGEvalResult:
    """
    评估单条 Case。

    注意：这里是"最小规则评估"，不是完美的自动打分。
    实际项目中建议：
      - 先跑自动规则筛出明显问题
      - 再人工过一遍边界 case
    """
    result = RAGEvalResult(case_id=case.case_id, model_output=answer)

    norm_answer = answer.replace(" ", "").replace("\n", "").lower()

    # ---- 幻觉检查：expected_absent 中任何词出现就算 hallucination ----
    for bad_word in case.expected_absent:
        if bad_word.replace(" ", "").lower() in norm_answer:
            result.hallucination = True
            result.extra_notes += f"出现禁用词: {bad_word}; "
            break

    # ---- 答非所问检查：expected_exact 中关键词一个都没命中 ----
    if case.expected_exact:
        hit_count = 0
        for good_word in case.expected_exact:
            if good_word.replace(" ", "").lower() in norm_answer:
                hit_count += 1
        if hit_count == 0:
            result.irrelevant = True
            result.extra_notes += "未命中任何期望关键词; "

    # ---- 拒答检查 ----
    refuse_words = ["知识库中没有找到", "未找到相关内容", "未明确提及"]
    answer_refused = any(w in answer for w in refuse_words)

    if case.should_refuse and not answer_refused:
        result.wrong_refuse = True
        result.extra_notes += "应该拒答但没有; "

    if not case.should_refuse and answer_refused and case.expected_exact:
        # 如果 expected_exact 非空（说明应该有答案）但模型拒答了，算错
        result.wrong_refuse = True
        result.extra_notes += "不该拒答但拒答了; "

    # ---- 引用检查（简单规则：没有参考来源就算不清晰）----
    if case.difficulty == "normal_hit":
        if not any(w in answer for w in ["参考来源", "参考文献", "[", "]"]):
            result.citation_messy = True
            result.extra_notes += "缺少引用标注; "

    return result


def eval_runner(all_cases: list[RAGEvalCase]):
    """
    跑完整评估集，输出汇总。
    """
    results: list[RAGEvalResult] = []
    for case in all_cases:
        print(f"\n===== 评估 {case.case_id} =====")
        print(f"问题: {case.question}")
        print(f"类型: {case.question_type} / 难度: {case.difficulty}")

        # 1. 调用 RAG
        answer = run_rag_query(case.question)
        print(f"模型输出:\n{answer}\n")

        # 2. 规则评估
        result = check_case(case, answer)
        results.append(result)

        # 3. 打印问题项
        issues = []
        if result.irrelevant:    issues.append("答非所问")
        if result.hallucination: issues.append("幻觉")
        if result.wrong_refuse:  issues.append("拒答错误")
        if result.citation_messy:issues.append("引用不清晰")
        if result.synthesis_bad: issues.append("整合很乱")

        if issues:
            print(f"❌ 问题: {', '.join(issues)}")
            if result.extra_notes:
                print(f"   细节: {result.extra_notes}")
        else:
            print("✅ 通过（规则层面）")

    # ===== 4. 汇总统计 =====
    total = len(results)
    pass_count = sum(
        1 for r in results
        if not any([r.irrelevant, r.hallucination, r.wrong_refuse, r.citation_messy, r.synthesis_bad])
    )

    print("\n" + "=" * 50)
    print(f"评估汇总：")
    print(f"  总 Case 数: {total}")
    print(f"  规则通过数: {pass_count}")
    print(f"  规则通过率: {pass_count/total*100:.1f}%")
    print()
    print(f"  答非所问数  : {sum(1 for r in results if r.irrelevant)}")
    print(f"  幻觉数      : {sum(1 for r in results if r.hallucination)}")
    print(f"  拒答错误数  : {sum(1 for r in results if r.wrong_refuse)}")
    print(f"  引用混乱数  : {sum(1 for r in results if r.citation_messy)}")

    # 可选：导出 JSON 方便后续对比
    export_data = [
        {
            "case_id": r.case_id,
            "output": r.model_output,
            "issues": {
                "irrelevant": r.irrelevant,
                "hallucination": r.hallucination,
                "wrong_refuse": r.wrong_refuse,
                "citation_messy": r.citation_messy,
            },
            "notes": r.extra_notes,
        }
        for r in results
    ]
    with open("rag_eval_result.json", "w", encoding="utf-8") as f:
        json.dump(export_data, f, ensure_ascii=False, indent=2)
    print("\n详细结果已导出到 rag_eval_result.json")
```
