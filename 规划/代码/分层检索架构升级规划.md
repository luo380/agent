## 架构总览

```
用户问题 → [L1 查询预处理] → [L2 结构化理解] → [L3 多路Rewrite] → [L4 混合召回] → [L5 语义重排] → [L6 规则控制]
```

```python
# ============================================================
# 【新增】结构化 Query Understanding 模块
# 作用：把用户问题解析为 question_type + entities + relation + bridge
# 这是多路 Rewrite 的基础，比原来的单一关键词提取更精细
# ============================================================

from dataclasses import dataclass, field
from typing import Optional


@dataclass
class QueryStructure:
    """
    结构化查询表示例：
    原始问题："我家扫地机器人能不能连5G WiFi？"
    
    question_type: yes_no
    entities: ["扫地机器人", "5G WiFi"]        # 核心实体（产品、功能、技术名词）
    relation: "连接/支持"                       # 实体间的关系动词
    bridge_terms: ["我家"]                      # 修饰词/口语词（后续会被过滤）
    normalized_query: "扫地机器人 连接 5G WiFi" # 标准化后的检索式
    """
    # 问题类型：yes_no / frequency / how_to / list / fact
    question_type: str
    
    # 核心实体列表（产品名、功能名、技术术语等）
    entities: list[str] = field(default_factory=list)
    
    # 关系词（支持、连接、兼容、包含等）
    relation: str = ""
    
    # 桥接词/修饰词（我家、请问、能不能等口语化成分）
    bridge_terms: list[str] = field(default_factory=list)
    
    # 标准化后的检索式（去除口语、保留核心语义）
    normalized_query: str = ""
    
    # 原始问题（保留备用）
    original_query: str = ""


# 关系动词词典：用于识别"实体A 关系 实体B"中的关系词
RELATION_VERBS = (
    # 支持/兼容类
    "支持", "兼容", "适用", "匹配", "可以用", "能用",
    # 连接/操作类
    "连接", "绑定", "配对", "接入", "添加", "设置", "配置",
    # 包含/组成类
    "包含", "包括", "有", "由...组成", "分为",
    # 控制/功能类
    "控制", "操控", "管理", "查看", "查询", "修改",
    # 状态/属性类
    "是", "属于", "需要", "必须", "应该",
)

# 实体后缀词：用于辅助识别实体边界
ENTITY_SUFFIXES = (
    # 产品类
    "机器人", "音箱", "摄像头", "门锁", "灯泡", "插座", "开关", "窗帘",
    # 技术类
    "WiFi", "蓝牙", "Zigbee", "NFC", "5G", "2.4G", "USB", "HDMI",
    # 功能类
    "功能", "模式", "场景", "定时", "自动化", "联动",
)

# 口语化桥接词（需要在标准化时过滤掉）
COLLOQUIAL_BRIDGE = (
    "我家", "家里的", "我这边", "我现在", "请问", "问一下", "麻烦问下",
    "能不能", "可以吗", "行不行", "好不好", "是不是", "有没有",
)


def _extract_entities(text: str) -> list[str]:
    """
    从文本中提取核心实体。
    
    策略：
    1. 先去掉关系动词、疑问词、口语词
    2. 按产品前缀、技术后缀切分
    3. 保留长度 >= 2 的候选词
    
    示例：
    输入："扫地机器人连接5G WiFi"
    输出：["扫地机器人", "5G WiFi"]
    """
    normalized = normalize_for_grounding(text)
    if not normalized:
        return []
    
    entities: list[str] = []
    
    # 第一步：去掉所有关系动词，露出实体骨架
    candidate = normalized
    for verb in RELATION_VERBS:
        candidate = candidate.replace(verb, " ")
    
    # 第二步：去掉疑问词和口语词
    for word in QUESTION_WORDS + YES_NO_ANCHORS + HOW_TO_ANCHORS + LIST_ANCHORS:
        candidate = candidate.replace(word, " ")
    for bridge in COLLOQUIAL_BRIDGE:
        candidate = candidate.replace(bridge, " ")
    
    # 第三步：按空格切分，过滤太短的词
    for token in candidate.split():
        token = token.strip()
        if len(token) >= 2:
            # 去掉产品前缀（如果实体太长）
            cleaned = _strip_product_prefix(token)
            if len(cleaned) >= 2 and cleaned not in entities:
                entities.append(cleaned)
    
    return entities


def _extract_relation(text: str) -> str:
    """
    从文本中提取关系动词。
    
    策略：
    1. 遍历 RELATION_VERBS，找第一个出现在文本中的关系词
    2. 如果有多个，优先选最长的那个（更具体）
    
    示例：
    输入："扫地机器人能不能连接5G WiFi"
    输出："连接"
    """
    normalized = normalize_for_grounding(text)
    if not normalized:
        return ""
    
    # 按长度降序排序，优先匹配长关系词（避免"支持"匹配到"不支持"）
    sorted_verbs = sorted(RELATION_VERBS, key=len, reverse=True)
    
    for verb in sorted_verbs:
        if verb in normalized:
            return verb
    
    return ""


def _extract_bridge_terms(text: str) -> list[str]:
    """
    提取桥接词/修饰词（口语化成分，后续检索时需要过滤）。
    
    示例：
    输入："我家扫地机器人能不能连5G WiFi"
    输出：["我家", "能不能"]
    """
    normalized = normalize_for_grounding(text)
    if not normalized:
        return []
    
    bridges: list[str] = []
    for bridge in COLLOQUIAL_BRIDGE:
        if bridge in normalized:
            bridges.append(bridge)
    
    return bridges


def build_normalized_query(text: str, entities: list[str], relation: str) -> str:
    """
    根据实体和关系构建标准化检索式。
    
    策略：
    1. 如果有实体 + 关系：拼接为 "实体1 关系 实体2"
    2. 如果只有实体：拼接为 "实体1 实体2"
    3. 如果都没有：返回原文本（兜底）
    
    示例：
    entities=["扫地机器人", "5G WiFi"], relation="连接"
    输出："扫地机器人 连接 5G WiFi"
    """
    if entities and relation:
        # 实体 + 关系 + 实体（如果有2个实体）
        if len(entities) >= 2:
            return f"{entities[0]} {relation} {entities[1]}"
        # 只有1个实体
        return f"{entities[0]} {relation}"
    
    if entities:
        # 只有实体，用空格拼接
        return " ".join(entities[:3])  # 最多3个实体，避免太长
    
    # 兜底：返回归一化后的原文本
    return normalize_for_grounding(text)


def understand_query(question: str) -> QueryStructure:
    """
    【核心函数】对用户问题进行结构化理解。
    
    返回 QueryStructure 包含：
    - question_type: 问题类型
    - entities: 核心实体列表
    - relation: 关系动词
    - bridge_terms: 口语化桥接词
    - normalized_query: 标准化检索式
    
    使用示例：
    >>> structure = understand_query("我家扫地机器人能不能连5G WiFi？")
    >>> structure.question_type
    'yes_no'
    >>> structure.entities
    ['扫地机器人', '5G WiFi']
    >>> structure.relation
    '连接'
    >>> structure.normalized_query
    '扫地机器人 连接 5G WiFi'
    """
    # 1. 识别问题类型（复用现有逻辑）
    question_type = infer_question_type(question)
    
    # 2. 提取实体
    entities = _extract_entities(question)
    
    # 3. 提取关系词
    relation = _extract_relation(question)
    
    # 4. 提取桥接词
    bridge_terms = _extract_bridge_terms(question)
    
    # 5. 构建标准化检索式
    normalized_query = build_normalized_query(question, entities, relation)
    
    return QueryStructure(
        question_type=question_type,
        entities=entities,
        relation=relation,
        bridge_terms=bridge_terms,
        normalized_query=normalized_query,
        original_query=question,
    )
```

## 文件 2：`core/service/query_rewrite.py` 新建

```python
"""
多路 Query Rewrite 模块

作用：基于结构化理解结果，生成多条"更像标准检索问法"的变体，
     再一起参与召回，提高检索覆盖率。

设计参考：
- Azure AI Search Query Rewrite
- Algolia Synonyms
- Elasticsearch Query Pipeline
"""

from __future__ import annotations

from dataclasses import dataclass, field
from core.service.rag_grounding import (
    understand_query,
    QueryStructure,
    normalize_for_grounding,
    extract_question_focus_terms,
)


@dataclass
class RewriteForm:
    """
    单条 Rewrite 变体。
    
    示例：
    text: "扫地机器人 连接 5G WiFi"
    strategy: "entity_relation"      # 生成策略
    weight: 1.0                      # 权重（用于后续召回加权）
    """
    # 改写后的查询文本
    text: str
    
    # 生成策略标识（用于调试和规则控制）
    strategy: str
    
    # 权重（原始问题=1.0，变体可能降低权重）
    weight: float = 1.0


# ============================================================
# 策略 1：实体 + 关系标准式
# 适用场景：有明确实体和关系的问题
# 示例："扫地机器人 连接 5G WiFi"
# ============================================================
def _rewrite_entity_relation(structure: QueryStructure) -> list[RewriteForm]:
    """基于实体+关系生成标准检索式。"""
    forms: list[RewriteForm] = []
    
    if not structure.entities:
        return forms
    
    # 主变体：实体1 + 关系 + 实体2
    if len(structure.entities) >= 2 and structure.relation:
        text = f"{structure.entities[0]} {structure.relation} {structure.entities[1]}"
        forms.append(RewriteForm(text=text, strategy="entity_relation", weight=1.0))
    
    # 变体：只有实体（去掉关系，兜底匹配）
    if len(structure.entities) >= 2:
        text = f"{structure.entities[0]} {structure.entities[1]}"
        forms.append(RewriteForm(text=text, strategy="entity_only", weight=0.8))
    
    # 变体：单个实体 + 关系（如果只有1个实体）
    if len(structure.entities) == 1 and structure.relation:
        text = f"{structure.entities[0]} {structure.relation}"
        forms.append(RewriteForm(text=text, strategy="single_entity_relation", weight=0.7))
    
    return forms


# ============================================================
# 策略 2：关系变体展开
# 适用场景：关系词有多个同义表达
# 示例："连接" → "支持" / "兼容" / "接入"
# ============================================================

# 关系同义词映射表（可配置化，后续可移到数据库/配置文件）
RELATION_SYNONYMS = {
    "连接": ["支持", "兼容", "接入", "配对", "绑定"],
    "支持": ["兼容", "可以用", "能用", "适用"],
    "兼容": ["支持", "适用", "匹配"],
    "控制": ["操控", "管理", "操作"],
    "包含": ["包括", "有", "分为", "由...组成"],
    "添加": ["绑定", "连接", "配置", "设置"],
}


def _rewrite_relation_variants(structure: QueryStructure) -> list[RewriteForm]:
    """基于关系同义词生成变体。"""
    forms: list[RewriteForm] = []
    
    if not structure.relation:
        return forms
    
    # 查找同义词
    synonyms = RELATION_SYNONYMS.get(structure.relation, [])
    if not synonyms:
        return forms
    
    # 为每个同义词生成变体
    for synonym in synonyms[:3]:  # 最多3个变体，避免爆炸
        if structure.entities:
            # 有实体：实体 + 同义关系
            if len(structure.entities) >= 2:
                text = f"{structure.entities[0]} {synonym} {structure.entities[1]}"
            else:
                text = f"{structure.entities[0]} {synonym}"
        else:
            # 无实体：只有关系词（较少见）
            text = synonym
        
        forms.append(RewriteForm(text=text, strategy="relation_synonym", weight=0.6))
    
    return forms


# ============================================================
# 策略 3：倒装/抽象变体
# 适用场景：知识库可能用不同语序描述同一件事
# 示例："5G WiFi 扫地机器人 兼容性"
# ============================================================
def _rewrite_inverted(structure: QueryStructure) -> list[RewriteForm]:
    """生成倒装/抽象变体。"""
    forms: list[RewriteForm] = []
    
    if len(structure.entities) < 2:
        return forms
    
    # 倒装：实体2 + 关系 + 实体1
    if structure.relation:
        text = f"{structure.entities[1]} {structure.relation} {structure.entities[0]}"
        forms.append(RewriteForm(text=text, strategy="inverted", weight=0.5))
    
    # 抽象：实体2 + 抽象名词（兼容性/支持情况/使用方法）
    abstract_suffixes = {
        "yes_no": ["兼容性", "支持情况"],
        "how_to": ["使用方法", "操作步骤", "设置方法"],
        "frequency": ["频率", "周期", "间隔时间"],
        "list": ["列表", "清单", "有哪些"],
    }
    
    suffixes = abstract_suffixes.get(structure.question_type, [])
    for suffix in suffixes[:2]:  # 最多2个抽象变体
        text = f"{structure.entities[1]} {suffix}"
        forms.append(RewriteForm(text=text, strategy="abstract", weight=0.4))
    
    return forms


# ============================================================
# 策略 4：关键词聚焦变体（复用现有逻辑）
# 适用场景：保留原有的 extract_question_focus_terms 能力
# ============================================================
def _rewrite_focus_terms(structure: QueryStructure) -> list[RewriteForm]:
    """基于原有关键词提取逻辑生成变体。"""
    forms: list[RewriteForm] = []
    
    # 复用现有的 extract_question_focus_terms
    terms = extract_question_focus_terms(structure.original_query)
    
    # 取长度 >= 4 的长关键词作为变体
    for term in terms:
        if len(term) >= 4:
            forms.append(RewriteForm(text=term, strategy="focus_term", weight=0.6))
    
    return forms[:3]  # 最多3个


# ============================================================
# 主入口函数
# ============================================================
def generate_rewrite_forms(
    question: str,
    *,
    max_forms: int = 8,           # 最多生成多少条变体
    min_length: int = 2,          # 变体最小长度
    max_length: int = 50,         # 变体最大长度
    whitelist: set[str] | None = None,   # 白名单（必须包含的词）
    blacklist: set[str] | None = None,   # 黑名单（不能包含的词）
) -> list[RewriteForm]:
    """
    【核心函数】为用户问题生成多条 Rewrite 变体。
    
    参数：
    - question: 用户原始问题
    - max_forms: 最多生成多少条变体（默认8条）
    - min_length: 变体最小长度（过滤太短的）
    - max_length: 变体最大长度（过滤太长的）
    - whitelist: 白名单集合，变体必须包含其中至少一个词
    - blacklist: 黑名单集合，变体不能包含其中任何词
    
    返回：
    - 按权重降序排列的 RewriteForm 列表
    
    使用示例：
    >>> forms = generate_rewrite_forms("我家扫地机器人能不能连5G WiFi？")
    >>> for form in forms:
    ...     print(f"{form.text} (策略={form.strategy}, 权重={form.weight})")
    
    输出示例：
    扫地机器人 连接 5G WiFi (策略=entity_relation, 权重=1.0)
    扫地机器人 5G WiFi (策略=entity_only, 权重=0.8)
    扫地机器人 支持 5G WiFi (策略=relation_synonym, 权重=0.6)
    扫地机器人 兼容 5G WiFi (策略=relation_synonym, 权重=0.6)
    5G WiFi 连接 扫地机器人 (策略=inverted, 权重=0.5)
    5G WiFi 兼容性 (策略=abstract, 权重=0.4)
    """
    # 第一步：结构化理解
    structure = understand_query(question)
    
    # 第二步：应用4种策略生成变体
    all_forms: list[RewriteForm] = []
    
    # 策略1：实体+关系标准式（权重最高）
    all_forms.extend(_rewrite_entity_relation(structure))
    
    # 策略2：关系同义词变体
    all_forms.extend(_rewrite_relation_variants(structure))
    
    # 策略3：倒装/抽象变体
    all_forms.extend(_rewrite_inverted(structure))
    
    # 策略4：关键词聚焦变体
    all_forms.extend(_rewrite_focus_terms(structure))
    
    # 第三步：去重（归一化后去重）
    seen: set[str] = set()
    unique_forms: list[RewriteForm] = []
    for form in all_forms:
        normalized = normalize_for_grounding(form.text)
        if normalized not in seen and len(normalized) >= min_length:
            seen.add(normalized)
            unique_forms.append(form)
    
    # 第四步：应用白名单/黑名单过滤
    filtered_forms: list[RewriteForm] = []
    for form in unique_forms:
        # 黑名单检查
        if blacklist:
            if any(bad in form.text for bad in blacklist):
                continue
        
        # 白名单检查（如果指定了白名单，变体必须包含至少一个白名单词）
        if whitelist:
            if not any(good in form.text for good in whitelist):
                continue
        
        # 长度检查
        if len(form.text) > max_length:
            continue
        
        filtered_forms.append(form)
    
    # 第五步：按权重降序排序，取 Top max_forms
    filtered_forms.sort(key=lambda f: f.weight, reverse=True)
    
    return filtered_forms[:max_forms]
```

```python
"""
规则控制层（Query Rules / Business Control Layer）

作用：纯模型不够，商业产品都会保留人工可控层：
- 同义词表管理
- 白名单/黑名单
- Boost/Bury（提权/降权）
- 重定向规则
- Query Pipeline

设计参考：
- Algolia Rules
- Coveo Query Pipeline
- Elasticsearch Query Pipeline
"""

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional


# ============================================================
# 规则 1：同义词表
# ============================================================
@dataclass
class SynonymRule:
    """
    同义词规则。
    
    示例：
    original: "WiFi"
    synonyms: ["无线网", "无线网络", "WIFI", "wi-fi"]
    
    效果：用户搜"WiFi"时，同时召回包含"无线网"的文档
    """
    # 原始词
    original: str
    
    # 同义词列表
    synonyms: list[str] = field(default_factory=list)
    
    # 是否启用
    enabled: bool = True


# 默认同义词表（后续可迁移到数据库/配置文件）
DEFAULT_SYNONYMS: list[SynonymRule] = [
    SynonymRule(
        original="WiFi",
        synonyms=["无线网", "无线网络", "WIFI", "wi-fi", "WLAN"],
    ),
    SynonymRule(
        original="蓝牙",
        synonyms=["bluetooth", "BT"],
    ),
    SynonymRule(
        original="扫地机器人",
        synonyms=["扫地机", "扫拖机器人", "清洁机器人", "吸尘器机器人"],
    ),
    SynonymRule(
        original="摄像头",
        synonyms=["监控", "摄像机", "监控摄像头"],
    ),
]


# ============================================================
# 规则 2：白名单/黑名单
# ============================================================
@dataclass
class QueryFilterRule:
    """
    查询过滤规则。
    
    白名单：查询必须包含这些词之一，否则拒绝检索（节省资源）
    黑名单：查询包含这些词时，直接过滤掉（避免无意义检索）
    """
    # 规则类型：whitelist / blacklist
    rule_type: str
    
    # 规则词列表
    terms: list[str] = field(default_factory=list)
    
    # 规则描述
    description: str = ""
    
    # 是否启用
    enabled: bool = True


# 默认黑名单（口语词、无意义词）
DEFAULT_BLACKLIST = QueryFilterRule(
    rule_type="blacklist",
    terms=[
        # 口语化语气词
        "请问", "问一下", "麻烦问下", "我想问", "帮我查",
        # 无意义填充词
        "这个", "那个", "什么", "咋", "咋样",
        # 敏感/不相关词

# 规则引擎主类
# ============================================================
class QueryRuleEngine:
    """
    查询规则引擎。
    
    作用：在检索前应用所有规则，对查询进行预处理和约束。
    
    使用示例：
    >>> engine = QueryRuleEngine()
    >>> processed = engine.apply_rules("我家扫地机器人能不能连5G WiFi？")
    >>> print(processed.query)
    '扫地机器人 连接 5G WiFi'
    >>> print(processed.is_blocked)
    False
    """
    
    def __init__(
        self,
        synonyms: list[SynonymRule] | None = None,
        blacklist: QueryFilterRule | None = None,
        whitelist: QueryFilterRule | None = None,
        boost_bury_rules: list[BoostBuryRule] | None = None,
        redirect_rules: list[RedirectRule] | None = None,
    ):
        # 初始化规则（使用默认值或传入值）
        self.synonyms = synonyms or DEFAULT_SYNONYMS
        self.blacklist = blacklist or DEFAULT_BLACKLIST
        self.whitelist = whitelist or DEFAULT_WHITELIST
        self.boost_bury_rules = boost_bury_rules or DEFAULT_BOOST_BURY_RULES
        self.redirect_rules = redirect_rules or DEFAULT_REDIRECT_RULES
    
    @dataclass
    class ProcessedQuery:
        """规则处理后的查询结果。"""
        # 最终查询文本
        query: str
        
        # 是否被黑名单拦截
        is_blocked: bool = False
        
        # 是否通过白名单验证
        passes_whitelist: bool = True
        
        # 是否命中重定向
        redirected: bool = False
        
        # 原始查询（如果重定向了）
        original_query: str = ""
        
        # 扩展的同义词列表
        expanded_synonyms: list[str] = field(default_factory=list)
        
        # 应用的 boost/bury 规则
        applied_boost_bury: list[tuple[str, float]] = field(default_factory=list)
    
    def apply_rules(self, query: str) -> ProcessedQuery:
        """
        【核心函数】应用所有规则处理查询。
        
        处理顺序：
        1. 重定向检查（优先）
        2. 黑名单检查
        3. 白名单检查
        4. 同义词扩展
        5. Boost/Bury 规则匹配
        
        返回 ProcessedQuery 对象，包含处理结果和元信息。
        """
        result = self.ProcessedQuery(query=query, original_query=query)
        
        # 第一步：重定向检查
        redirect_result = self._apply_redirect(query)
        if redirect_result:
            result.query = redirect_result
            result.redirected = True
        
        # 第二步：黑名单检查
        if self._is_blacklisted(result.query):
            result.is_blocked = True
            return result
        
        # 第三步：白名单检查
        result.passes_whitelist = self._passes_whitelist(result.query)
        
        # 第四步：同义词扩展
        result.expanded_synonyms = self._expand_synonyms(result.query)
        
        # 第五步：Boost/Bury 规则匹配
        result.applied_boost_bury = self._match_boost_bury(result.query)
        
        return result
    
    def _apply_redirect(self, query: str) -> Optional[str]:
        """应用重定向规则。"""
        normalized_query = query.lower().strip()
        
        for rule in self.redirect_rules:
            if not rule.enabled:
                continue
            
            if rule.exact_match:
                # 精确匹配
                if normalized_query == rule.trigger.lower():
                    return rule.redirect_to
            else:
                # 包含匹配
                if rule.trigger.lower() in normalized_query:
                    return rule.redirect_to
        
        return None
    
    def _is_blacklisted(self, query: str) -> bool:
        """检查是否命中黑名单。"""
        if not self.blacklist.enabled:
            return False
        
        normalized_query = query.lower()
        return any(term.lower() in normalized_query for term in self.blacklist.terms)
    
    def _passes_whitelist(self, query: str) -> bool:
        """检查是否通过白名单验证。"""
        if not self.whitelist.enabled:
            return True  # 未启用白名单则默认通过
        
        normalized_query = query.lower()
        return any(term.lower() in normalized_query for term in self.whitelist.terms)
    
    def _expand_synonyms(self, query: str) -> list[str]:
        """扩展同义词。"""
        expanded: list[str] = []
        normalized_query = query.lower()
        
        for rule in self.synonyms:
            if not rule.enabled:
                continue
            
            # 检查原始词是否在查询中
            if rule.original.lower() in normalized_query:
                expanded.extend(rule.synonyms)
        
        return expanded
    
    def _match_boost_bury(self, query: str) -> list[tuple[str, float]]:
        """匹配 Boost/Bury 规则。"""
        matched: list[tuple[str, float]] = []
        normalized_query = query.lower()
        
        for rule in self.boost_bury_rules:
            if not rule.enabled:
                continue
            
            if rule.term.lower() in normalized_query:
                matched.append((rule.term, rule.boost_factor))
        
        return matched
    
    def apply_boost_to_score(self, score: float, chunk_text: str) -> float:
        """
        对检索结果分数应用 Boost/Bury 调整。
        
        参数：
        - score: 原始分数
        - chunk_text: 知识块文本（用于匹配 boost/bury 规则）
        
        返回：
        - 调整后的分数
        """
        adjusted_score = score
        normalized_text = chunk_text.lower()
        
        for rule in self.boost_bury_rules:
            if not rule.enabled:
                continue
            
            if rule.term.lower() in normalized_text:
                adjusted_score *= rule.boost_factor
        
        return adjusted_score
```

```python
# ============================================================
# 【新增】多路 Rewrite 召回入口
# 作用：基于 Rewrite 变体列表，并行执行多路召回，再融合结果
# ============================================================
def hybrid_search_with_rewrite(
    db: Session,
    *,
    user_id: int,
    query_text: str,
    query_embedding: list[float],
    document_ids: Sequence[int] | None = None,
    top_k: int = 10,
    vector_weight: float = 0.65,
    bm25_weight: float = 0.35,
    rule_engine: QueryRuleEngine | None = None,  # 【新增】规则引擎
) -> list[RetrievedChunk]:
    """
    【核心函数】带 Rewrite 和规则控制的混合检索。
    
    检索流程：
    1. 应用规则引擎处理查询（重定向、黑白名单、同义词）
    2. 生成多条 Rewrite 变体
    3. 对每个变体执行 hybrid_search（并行或串行）
    4. 融合所有变体的结果，按权重加权排序
    
    参数：
    - rule_engine: 查询规则引擎（可选，不传则使用默认规则）
    
    返回：
    - 融合后的检索结果列表（按 final_score 降序）
    """
    # 第一步：应用规则引擎
    engine = rule_engine or QueryRuleEngine()
    processed = engine.apply_rules(query_text)
    
    # 黑名单拦截：直接返回空
    if processed.is_blocked:
        return []
    
    # 白名单未通过：降级为宽松模式（只返回高置信结果）
    if not processed.passes_whitelist:
        # 可以选择返回空或降低 top_k
        top_k = max(1, top_k // 2)
    
    # 第二步：生成 Rewrite 变体
    rewrite_forms = generate_rewrite_forms(
        query_text,
        whitelist=set(engine.whitelist.terms) if engine.whitelist.enabled else None,
        blacklist=set(engine.blacklist.terms) if engine.blacklist.enabled else None,
    )
    
    # 第三步：对每个变体执行召回（这里简化为串行，生产环境可并行）
    all_results: dict[int, RetrievedChunk] = {}
    
    # 3.1 原始查询召回（权重最高）
    original_results = hybrid_search(
        db,
        user_id=user_id,
        query_text=query_text,
        query_embedding=query_embedding,
        document_ids=document_ids,
        top_k=top_k,
        vector_weight=vector_weight,
        bm25_weight=bm25_weight,
    )
    for chunk in original_results:
        all_results[chunk.chunk_id] = chunk
    
    # 3.2 Rewrite 变体召回
    for form in rewrite_forms:
        # 注意：Rewrite 变体没有 embedding，这里需要重新计算
        # 简化处理：只用 BM25 召回（生产环境应调用 embedding 服务）
        from core.service.embedding import embed_text
        try:
            form_embedding = embed_text(form.text)
        except Exception:
            form_embedding = query_embedding  # 降级：用原查询的 embedding
        
        form_results = hybrid_search(
            db,
            user_id=user_id,
            query_text=form.text,
            query_embedding=form_embedding,
            document_ids=document_ids,
            top_k=max(1, top_k // 2),  # 每个变体召回数量减半
            vector_weight=vector_weight,
            bm25_weight=bm25_weight,
        )
        
        # 融合结果：如果 chunk 已存在，取最高分；否则新增
        for chunk in form_results:
            # 应用 Rewrite 权重
            chunk.final_score *= form.weight
            
            if chunk.chunk_id in all_results:
                # 已存在：取最高分
                existing = all_results[chunk.chunk_id]
                if chunk.final_score > existing.final_score:
                    all_results[chunk.chunk_id] = chunk
            else:
                all_results[chunk.chunk_id] = chunk
    
    # 第四步：应用 Boost/Bury 规则调整分数
    if engine.boost_bury_rules:
        for chunk_id, chunk in all_results.items():
            chunk.final_score = engine.apply_boost_to_score(
                chunk.final_score,
                chunk.content,
            )
    
    # 第五步：排序并取 Top K
    sorted_results = sorted(
        all_results.values(),
        key=lambda c: c.final_score,
        reverse=True,
    )
    
    return sorted_results[:top_k]
```

```python
# ============================================================
# 【新增】行为反馈接口（预留）
# 作用：记录用户点击、停留、后续成功率，用于后续学习优化

from dataclasses import dataclass, field
from datetime import datetime


@dataclass
class QueryFeedback:
    """
    查询行为反馈记录。
    
    示例：
    query: "扫地机器人 连接 5G WiFi"
    clicked_chunk_ids: [101, 105]        # 用户点击了哪些结果
    dwell_times: {101: 15.2, 105: 3.1}   # 在每个结果上停留的秒数
    success: True                         # 最终是否解决问题
    """
    # 查询文本
    query: str
    
    # 用户点击的 chunk_id 列表
    clicked_chunk_ids: list[int] = field(default_factory=list)
    
    # 每个 chunk 的停留时间（秒）
    dwell_times: dict[int, float] = field(default_factory=dict)
    
    # 是否最终解决问题
    success: bool | None = None
    
    # 查询时间戳
    timestamp: datetime = field(default_factory=datetime.now)


# 内存存储反馈记录（生产环境应写入数据库）
_feedback_log: list[QueryFeedback] = []


def log_query_feedback(feedback: QueryFeedback) -> None:
    """
    记录查询行为反馈。
    
    使用示例：
    >>> feedback = QueryFeedback(
    ...     query="扫地机器人 连接 5G WiFi",
    ...     clicked_chunk_ids=[101, 105],
    ...     dwell_times={101: 15.2, 105: 3.1},
    ...     success=True,
    ... )
    >>> log_query_feedback(feedback)
    """
    _feedback_log.append(feedback)


def get_top_clicked_chunks(limit: int = 10) -> dict[int, int]:
    """
    获取点击次数最多的 chunk 统计。
    
    返回：
    - {chunk_id: click_count} 字典，按点击次数降序
    
    用途：
    - 分析哪些结果最受欢迎
    - 后续可用于调整 boost/bury 规则
    """
    click_counter: dict[int, int] = {}
    
    for feedback in _feedback_log:
        for chunk_id in feedback.clicked_chunk_ids:
            click_counter[chunk_id] = click_counter.get(chunk_id, 0) + 1
    
    # 按点击次数降序
    sorted_chunks = sorted(click_counter.items(), key=lambda x: x[1], reverse=True)
    
    return dict(sorted_chunks[:limit])


def get_success_rate(query_pattern: str | None = None) -> float:
    """
    计算查询成功率。
    
    参数：
    - query_pattern: 可选，只统计包含该模式的查询
    
    返回：
    - 成功率（0.0 ~ 1.0）
    """
    if not _feedback_log:
        return 0.0
    
    filtered = _feedback_log
    if query_pattern:
        filtered = [f for f in _feedback_log if query_pattern.lower() in f.query.lower()]
    
    # 只统计有成功/失败标记的记录
    valid = [f for f in filtered if f.success is not None]
    if not valid:
        return 0.0
    
    success_count = sum(1 for f in valid if f.success)
    return success_count / len(valid)
```

"""
完整分层检索链路使用示例。

流程：
用户问题 → 规则引擎 → 结构化理解 → 多路Rewrite → 混合召回 → 语义重排 → 返回结果

from core.service.retrieval import hybrid_search_with_rewrite
from core.service.query_rules import QueryRuleEngine
from core.service.embedding import embed_text
from core.db.session import get_db


def ask_knowledge_base(question: str, user_id: int):
    """
    完整的分层检索入口。
    
    示例：
    >>> answer = ask_knowledge_base("我家扫地机器人能不能连5G WiFi？", user_id=1)
    """

    rule_engine = QueryRuleEngine()

    db = next(get_db())

            query_embedding=query_embedding,
            top_k=5,
            rule_engine=rule_engine,
        )
        
        # 第四步：返回结果（给上层 rag.py 去拼上下文）

    finally:
        db.close()
