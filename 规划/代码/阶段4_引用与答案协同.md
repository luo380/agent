# 阶段4：引用与答案协同

## 核心目标
让引用不仅"存在"，而且"和答案对得上"。

## 涉及函数
- `build_citations_from_documents(...)` - 去重，不暴露内部标识
- `ensure_answer_has_document_citations(...)` - 引用兜底更精细
- `_build_reference_line(...)` - 用户可读，不暴露 page_6 占位符

## 4.1 _build_reference_line 用户可读性优化

### 原代码问题
1. 输出格式过于啰嗦，每一项都加前缀
2. "章节/区域：xxx" 这种机器感重
3. 没有把 document_name 和 source_page 做自然组合

### 优化后代码
```python
# ============================================================
# 优化后：_build_reference_line
# 改动点：
#   1. 输出更自然，去掉"章节/区域："这类前缀
#   2. document_name 简化处理（去掉过长路径）
#   3. 空值不展示，避免出现 "；；" 这种拼接痕迹
# ============================================================

def _shorten_document_name(name: str, max_len: int = 40) -> str:
    """
    文档名过长时做简化展示。

    例：
      "XX公司扫地机器人S50_产品说明书_V3.2_2024年发布.pdf"
      → "XX公司扫地机器人S50...说明书.pdf"
    """
    name = name or ""
    if len(name) <= max_len:
        return name

    # 前后各留一部分，中间用 ... 代替
    keep_head = max_len // 2
    keep_tail = max_len - keep_head - 3  # 3 是 "..." 的长度
    return name[:keep_head] + "..." + name[-keep_tail:]


def _build_reference_line(
    document_name: str,
    source_page: Any,
    source_section: Any,
) -> str:
    """
    构建一条用户可读的引用行。

    【输出规则示例】
      只有文档名：
          产品说明书.pdf
      文档名 + 页码：
          产品说明书.pdf · 第5页
      文档名 + 页码 + 章节：
          产品说明书.pdf · 第5页 · 安装步骤
      文档名 + 章节（无页码，比如 DOCX）：
          产品说明书.pdf · 安装步骤

    【设计原则】
    - 用户可读优先，不暴露 page_6 / slide_3 这种内部结构化占位符
      （这些已经在 _format_user_facing_section 中做了转换）
    - 用 " · " 分隔，比 "；" 更自然
    - 空值直接跳过，绝不出现空字段的占位
    """
    page_number = _normalize_source_page(source_page)
    section_text = _format_user_facing_section(source_page, source_section)

    # 组装 parts，只保留有值的部分
    parts: list[str] = []

    # 1. 文档名（必选）
    parts.append(_shorten_document_name(document_name or "未知文档"))

    # 2. 页码（可选）
    if page_number is not None:
        parts.append(f"第{page_number}页")

    # 3. 章节/区域（可选）
    if section_text:
        parts.append(section_text)

    # 用中点连接，视觉更干净
    return " · ".join(parts)
```

## 4.2 build_citations_from_documents 去重

### 原代码问题
1. 同一文档同一页被多个 chunk 命中时，会在 citations 里出现多条几乎一样的记录
2. 前端展示引用列表时显得很臃肿

### 优化后代码
```python
# ============================================================
# 优化后：build_citations_from_documents
# 改动点：
#   1. 同文档+同页+同章节 做去重（去重后只保留 score 最高的那条）
#   2. content 字段做合理截断 + 去 HTML 标签
#   3. source_section 用用户可读格式，不暴露内部占位符
# ============================================================

def build_citations_from_documents(documents: Sequence[Document]) -> list[dict]:
    """
    从 Document.metadata 里整理 citations（给前端展示用）。

    【去重策略】
    去重 key = (document_id, source_page, source_section用户可读文本)
    - 同一文档的同一页同一章节，只保留 score 最高的那条
    - 这样前端展示的引用列表不会一堆重复的"第5页""第5页""第5页"

    【字段说明】
    - document_id / document_name：文档标识
    - source_page / source_section：用户可读的定位
    - score：用于前端排序或展示相关度
    - content：该 chunk 的代表性内容片段（最多300字）
    - chunk_count：合并了多少个 chunk（前端可以显示"含3段相关内容"）
    """
    if not documents:
        return []

    # ---------- 第一步：按去重 key 分组，保留每组最高分 ----------
    dedup_map: dict[tuple, dict] = {}

    for doc in documents:
        metadata = doc.metadata

        # 计算用户可读的 section（不暴露内部占位符）
        source_page = metadata.get("source_page")
        user_section = _format_user_facing_section(
            source_page,
            metadata.get("source_section"),
        )

        # 去重 key：文档ID + 页码 + 用户可读章节
        dedup_key = (
            metadata.get("document_id"),
            _normalize_source_page(source_page),
            user_section,
        )

        score = float(metadata.get("score") or metadata.get("final_score") or 0.0)

        # 准备 citation 条目
        raw_content = doc.page_content.strip()
        # 简单去多余空白（保留换行但去掉连续空行）
        clean_content = re.sub(r"\n{3,}", "\n\n", raw_content)

        entry = {
            "document_id": metadata.get("document_id"),
            "document_name": metadata.get("document_name", "未知文档"),
            "chunk_id": metadata.get("chunk_id"),
            "chunk_index": metadata.get("chunk_index"),
            "source_page": _normalize_source_page(source_page),
            "source_section": user_section,
            "score": score,
            "content": clean_content[:300],
            "_chunk_count": 1,  # 内部计数，最后再转成外部字段
        }

        # 如果 key 已存在，只保留 score 更高的
        if dedup_key in dedup_map:
            existing = dedup_map[dedup_key]
            if score > existing["score"]:
                # 新的分数更高，替换内容，但累加 chunk_count
                entry["_chunk_count"] = existing["_chunk_count"] + 1
                dedup_map[dedup_key] = entry
            else:
                # 已有条目分数更高，只累加 chunk_count
                existing["_chunk_count"] += 1
        else:
            dedup_map[dedup_key] = entry

    # ---------- 第二步：按 score 从高到低排序，输出最终列表 ----------
    sorted_entries = sorted(
        dedup_map.values(),
        key=lambda x: x["score"],
        reverse=True,
    )

    # 把内部的 _chunk_count 转成外部字段，并移除内部字段
    citations: list[dict] = []
    for entry in sorted_entries:
        chunk_count = entry.pop("_chunk_count")
        entry["chunk_count"] = chunk_count
        citations.append(entry)

    return citations
```

## 4.3 ensure_answer_has_document_citations 精细兜底

### 原代码问题
1. 答案末尾统一追加"参考来源："块，比较粗暴
2. 答案里如果已经有零星的 [1][2] 但不完整，没法智能补
3. 重复引用检查是按 (document_id, name, page, section) 做的，但和答案内容对不上

### 优化后代码
```python
# ============================================================
# 优化后：ensure_answer_has_document_citations
# 改动点：
#   1. 先检查答案内容里是否已经包含了引用编号
#   2. 如果已有引用，只补答案中实际出现过的编号
#   3. 如果答案完全没有引用，再做完整兜底
#   4. 引用去重按"文档+页码"粒度（比 build_citations 粒度更粗，避免列表过长）
# ============================================================

# 匹配答案中出现的 [数字] 引用编号
CITE_NUMBER_RE = re.compile(r"\[(\d+)\]")

# 匹配答案中是否已经有"参考来源"这种引用块的关键词
EXISTING_REF_BLOCK_RE = re.compile(r"(参考来源|参考文献|引用来源|资料来源)\s*[：:]", re.MULTILINE)


def _extract_cited_numbers(answer_text: str) -> list[int]:
    """
    从答案文本中提取出用户（或模型）实际引用过的编号。

    例：
      "支持WiFi连接 [1]，蓝牙配对参考 [3]"
      → [1, 3]
    """
    return sorted({
        int(m) for m in CITE_NUMBER_RE.findall(answer_text or "")
        if m.isdigit()
    })


def ensure_answer_has_document_citations(
    answer_text: str,
    documents: Sequence[Document],
) -> str:
    """
    给最终答案补引用兜底。

    【兜底策略（分三档）】
    档1：答案里已经有"参考来源"之类的完整块 → 不处理，直接返回
    档2：答案里有 [1][2] 之类的引用编号但没有来源块 → 只补出现过的编号
    档3：答案完全没有引用 → 在末尾统一追加完整参考来源列表

    【为什么分档？】
    - 档1说明模型已经自己处理好了引用，不要乱加
    - 档2是最常见的：模型标了 [1][3] 但没列来源，这时不能把 [2][4] 也补进去
    - 档3是兜底，保证用户至少能看到引用了哪些文档
    """
    clean_answer = (answer_text or "").strip()

    # 没有文档 → 没有引用可补
    if not documents:
        return clean_answer

    # 空答案兜底
    if not clean_answer:
        clean_answer = "我根据知识库整理了相关信息。"

    # ========== 档1：已经有完整的参考来源块 ==========
    if EXISTING_REF_BLOCK_RE.search(clean_answer):
        return clean_answer

    # ========== 先构建去重后的完整参考列表 ==========
    # 引用展示的去重粒度：(document_id, source_page)
    # 比 build_citations 的去重粒度更粗，防止末尾列表太长
    reference_lines: list[str] = []
    seen_rough_keys: set[tuple] = set()

    for doc in documents:
        metadata = doc.metadata
        document_name = metadata.get("document_name", "未知文档")
        source_page = metadata.get("source_page")
        section_text = _format_user_facing_section(
            source_page,
            metadata.get("source_section"),
        )

        rough_key = (
            metadata.get("document_id"),
            _normalize_source_page(source_page),
        )
        if rough_key in seen_rough_keys:
            continue
        seen_rough_keys.add(rough_key)

        reference_lines.append(
            _build_reference_line(document_name, source_page, section_text)
        )

    # 没有可用引用
    if not reference_lines:
        return clean_answer

    # ========== 档2：答案里有 [1][2] 编号，只补出现过的编号 ==========
    cited_numbers = _extract_cited_numbers(clean_answer)
    if cited_numbers:
        # 过滤：只保留编号在有效范围内的
        valid_refs = []
        for num in cited_numbers:
            if 1 <= num <= len(reference_lines):
                valid_refs.append(f"[{num}] {reference_lines[num - 1]}")

        if valid_refs:
            return (
                f"{clean_answer}\n\n"
                f"参考来源：\n"
                + "\n".join(valid_refs)
            )

    # ========== 档3：答案完全没有引用，追加完整参考列表 ==========
    numbered_lines = [
        f"[{i+1}] {line}"
        for i, line in enumerate(reference_lines)
    ]
    return (
        f"{clean_answer}\n\n"
        f"参考来源：\n"
        + "\n".join(numbered_lines)
    )
```
